cmake_minimum_required(VERSION 3.16)
project(tssm_fftest C)
find_path(FFMPEG_INCLUDE_DIR libavcodec/avcodec.h)
foreach(l avformat avcodec swscale swresample avutil)
  find_library(FF_${l} ${l})
  list(APPEND FFLIBS ${FF_${l}})
endforeach()
message(STATUS "ffmpeg: ${FFMPEG_INCLUDE_DIR} ${FFLIBS}")
add_executable(fftest main.c)
target_include_directories(fftest PRIVATE ${FFMPEG_INCLUDE_DIR})
target_link_libraries(fftest PRIVATE ${FFLIBS})
nx_generate_nacp(fftest.nacp NAME "TSSM FFmpeg test" AUTHOR "TSSM port" VERSION "0.0.1")
nx_create_nro(fftest NACP fftest.nacp)
